// CORE LISTEN-SPEAK LOOP (jarvis_core.py)
import pvporcupine, pyaudio, struct, wave
from faster_whisper import WhisperModel
import subprocess, tempfile

model = WhisperModel("large-v3", device="cuda")  # or "cpu"

def listen_for_wake_word():
    porcupine = pvporcupine.create(
        access_key="YOUR_PICOVOICE_KEY",
        keyword_paths=["hey-jarvis.ppn"]
    )
    pa = pyaudio.PyAudio()
    stream = pa.open(rate=porcupine.sample_rate, channels=1,
                     format=pyaudio.paInt16, input=True,
                     frames_per_buffer=porcupine.frame_length)
    while True:
        pcm = struct.unpack_from("h" * porcupine.frame_length,
                                  stream.read(porcupine.frame_length))
        if porcupine.process(pcm) >= 0:
            speak("Yes?")
            return record_command()

def transcribe(audio_path):
    segments, _ = model.transcribe(audio_path)
    return " ".join([s.text for s in segments])

def speak(text):
    subprocess.run(["piper", "--model", "en_US-lessac-high.onnx",
                    "--output_raw"], input=text.encode(), ...)

// VECTOR MEMORY SYSTEM
import chromadb
from sentence_transformers import SentenceTransformer

embed_model = SentenceTransformer("all-MiniLM-L6-v2")
client = chromadb.PersistentClient(path="./jarvis_memory")
collection = client.get_or_create_collection("conversations")

def remember(text, metadata={}):
    embedding = embed_model.encode(text).tolist()
    collection.add(embeddings=[embedding],
                   documents=[text],
                   ids=[str(time.time())],
                   metadatas=[metadata])

def recall(query, n=5):
    q_embed = embed_model.encode(query).tolist()
    results = collection.query(query_embeddings=[q_embed], n_results=n)
    return results["documents"][0]  # top relevant memories

    "notepad": "notepad.exe",
    "vscode": "code",
    "spotify": "C:/Users/.../Spotify.exe"

def open_app(name):
    path = find_nearest_app(name, APP_MAP)  # vector match
    subprocess.Popen([path])

def paste_prompt_file(filename):
    content = Path(f"./prompts/{filename}.txt").read_text()
    pyperclip.copy(content)
    pyautogui.hotkey("ctrl", "v")

def type_text(text, interval=0.03):
    pyautogui.typewrite(text, interval=interval)

def execute_action(intent: dict):
    if intent["type"] == "OPEN_APP":
        open_app(intent["app"])
    elif intent["type"] == "PASTE_PROMPT":
        paste_prompt_file(intent["file"])
    elif intent["type"] == "CLOSE_APP":
        subprocess.run(["taskkill", "/IM", intent["process"], "/F"])

JARVIS understands your screen in real-time
INTERMEDIATE ▾

OCR fallback: use pytesseract for ultra-fast text extraction from screen without needing a vision model.
// SCREEN CAPTURE + VISION
import mss, base64
from PIL import Image
import io, anthropic

def capture_screen():
    with mss.mss() as sct:
        screenshot = sct.grab(sct.monitors[1])
        img = Image.frombytes("RGB", screenshot.size, screenshot.bgra, "raw", "BGRX")
        buffer = io.BytesIO()
        img.save(buffer, format="JPEG", quality=85)
        return base64.b64encode(buffer.getvalue()).decode()

def ask_about_screen(question="What is on the screen?"):
    b64 = capture_screen()
    client = anthropic.Anthropic()
    response = client.messages.create(
        model="claude-opus-4-20250514",
        max_tokens=1024,
        messages=[{
            "role": "user",
            "content": [
                {"type": "image", "source": {"type": "base64",
                  "media_type": "image/jpeg", "data": b64}},
                {"type": "text", "text": question}
            ]
        }]
    )
    return response.content[0].text
